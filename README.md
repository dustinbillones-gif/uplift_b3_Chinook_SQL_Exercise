-- 4. Write a query that returns `SupportRepId` and the number of customers assigned to each rep — name the count column `NumCustomers` — sorted by `NumCustomers` in descending order
SELECT 
	c.SupportRepId,
	COUNT(c.CustomerId) AS NumCustomers
FROM customers c
GROUP BY c.SupportRepId
ORDER BY NumCustomers DESC;


-- 5. Write a query that returns `GenreId` and the average `UnitPrice` of tracks in that genre — name the average column `AvgPrice` — sorted by `AvgPrice` in descending order, limited to the top 5 rows.
SELECT * FROM tracks t -- for checking metadata
LIMIT 5

SELECT
	t.GenreId,
	AVG(t.UnitPrice) AS AvgPrice
FROM tracks t
GROUP BY GenreId
ORDER BY AvgPrice DESC 
LIMIT 5;

-- 1. Write a query that returns each genre's `Name` and the number of tracks in that genre — name the count column `NumTracks` — sorted by `NumTracks` in descending order, limited to the top 5 rows.
SELECT * FROM genres g LIMIT 5 -- for checking metadata

SELECT * FROM tracks t LIMIT 5 -- for checking metadata

SELECT
	g.Name AS Genre,
	COUNT(t.GenreId) AS NumTracks
FROM genres g
LEFT JOIN tracks t
	ON g.GenreId = t.GenreId
GROUP BY g.Name
ORDER BY NumTracks DESC
LIMIT 5;


-- 2. Write a query that returns the track name, album title, and artist name for every track — name these columns `TrackName`, `AlbumTitle`, and `ArtistName` respectively — limited to the first 10 rows.
SELECT * FROM tracks t LIMIT 5 -- for checking metadata

SELECT * FROM albums a LIMIT 5 -- for checking metadata

SELECT * FROM artists a LIMIT 5 -- for checking metadata

SELECT
	t.name AS TrackName,
	a.Title AS AlbumTitle,
	a2.Name AS ArtistName
FROM tracks t
LEFT JOIN albums a
	ON t.AlbumId = a.AlbumId
LEFT JOIN artists a2
	ON a.ArtistId = a2.ArtistId
LIMIT 10;


-- 3. Write a query that returns `FirstName`, `LastName`, the track name (named `TrackName`), `Quantity`, and `UnitPrice` from every invoice line ever billed to the customer with `CustomerId = 5`, sorted by `InvoiceLineId` in ascending order.
SELECT * FROM customers c -- LIMIT 5 -- for checking metadata

SELECT * FROM tracks t LIMIT 5 -- for checking metadata

SELECT * FROM invoices i -- LIMIT 5 -- for checking metadata

SELECT * FROM invoice_items ii LIMIT 5 -- for checking metadata


SELECT
	c.CustomerId,
	c.FirstName,
	c.LastName,
	t.Name AS TrackName,
	ii.Quantity,
	ii.UnitPrice
FROM customers c
LEFT JOIN invoices i
	ON c.CustomerId = i.CustomerId
LEFT JOIN invoice_items ii
	ON i.InvoiceId = ii.InvoiceId
LEFT JOIN tracks t
	ON t.TrackId = ii.TrackId
WHERE c.CustomerId = 5
ORDER BY ii.InvoiceLineId;


-- 4. Write a query that returns the `Name` of each playlist and the number of tracks it contains — name the count column `NumTracks` — sorted by `NumTracks` in descending order.
SELECT * FROM playlists p LIMIT 5 -- for checking metadata

SELECT * FROM tracks t LIMIT 5 -- for checking metadata

SELECT * FROM playlist_track pt LIMIT 5 -- for checking metadata

SELECT
	p.Name AS PlaylistName,
	COUNT(t.TrackId) AS NumTracks
FROM playlist_track pt
LEFT JOIN playlists p
	ON pt.PlaylistId = p.PlaylistId
LEFT JOIN tracks t
	ON pt.TrackId = t.TrackId
GROUP BY p.Name
ORDER BY NumTracks DESC;


-- 5. Write a query that returns `Country` and total revenue generated by customers in that country — computed as `SUM(UnitPrice * Quantity)` from the invoice line items, named `Revenue` — sorted by `Revenue` in descending order, limited to the top 5 rows.
SELECT * FROM invoice_items ii LIMIT 5 -- for checking metadata

SELECT * FROM customers c LIMIT 5 -- for checking metadata

SELECT * FROM invoices i LIMIT 5 -- for checking metadata

SELECT 
	c.Country,
	SUM(ii.UnitPrice * ii.Quantity) AS Revenue
FROM invoice_items ii
LEFT JOIN invoices i
	ON ii.InvoiceId = i.InvoiceId
LEFT JOIN customers c
	ON i.CustomerId = c.CustomerId
GROUP BY c.Country
ORDER BY Revenue DESC 
LIMIT 5;
